#!/usr/bin/env python3
"""Compact observation of selected original saved-owner PAGE replay, never native input."""
import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parent
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()

def compact(frame):
    s = frame['state']
    return {
        'stage': frame['stage'],
        'checked_voice_object': frame['checked_voice_object'],
        'system': s['system'], 'stack_object': s['stack_object'],
        'number': s['id'], 'special': s['special'], 'implicit': s['implicit'],
        'expected': s['expected'], 'actual': s['actual'], 'excess': s['excess'],
        'identity_reference_sha256': frame['identity_reference_sha256'],
        'actual_owners_slots_relations_and_manual_status_retained': frame['actual_owners_slots_relations_and_manual_status_retained'],
        'peer_measures': [{
            'part': m['part'], 'measure_object': m['measure_object'],
            'stack_object': m['stack_object'], 'actual_stack_same': m['actual_stack_same'],
            'expected': m['shared_expected'], 'actual': m['shared_actual'],
            'excess': m['shared_excess'], 'abnormal': m['abnormal'],
            'voices': [{k: v[k] for k in ('id', 'object', 'kind', 'starting_staff', 'chord_ids', 'chord_objects', 'termination', 'excess')}
                       for v in m['voices']]
        } for m in s['measures']],
    }

source = json.loads((ROOT / 'observed.json').read_text())
provenance = json.loads((ROOT / 'PROVENANCE.json').read_text())
assert provenance['exit_code'] == 0
assert provenance['book_before_sha256'] == provenance['book_after_sha256']
assert provenance['output'] == sha(ROOT / 'observed.json')
rows = []
for selected_index, replay in zip((1, 3), source['replays'], strict=True):
    frames = [compact(f) for f in replay['frames']]
    assert len({f['identity_reference_sha256'] for f in frames}) == 1
    for f in frames:
        assert f['actual_owners_slots_relations_and_manual_status_retained']
        for m in f['peer_measures']:
            assert m['actual_stack_same'] and m['stack_object'] == f['stack_object']
            for key in ('expected', 'actual', 'excess'):
                assert m[key] == f[key]
    rows.append({
        'system': frames[0]['system'], 'stack_index': selected_index,
        'stack_index_source': 'Direct selected getStacks().get(0) / get(2) in observer, one-based here',
        'common_termination': replay['common_termination'],
        'isPickup': replay['isPickup'],
        'selected_pickup_branch_executed': replay['selected_pickup_branch_executed'],
        'frames': frames,
    })
controls = []
for c in source['declared_controls']['cases']:
    b, a = c['before'], c['after']
    assert b['identity_reference_sha256'] == a['identity_reference_sha256']
    def checked(f):
        for m in f['state']['measures']:
            for v in m['voices']:
                if v['object'] == c['voice_object']:
                    ch = next(ch for ch in m['chords'] if ch['object'] == c['last_chord_object'])
                    return {'part': m['part'], 'voice_id': v['id'], 'voice_object': v['object'],
                            'last_chord_id': ch['id'], 'last_chord_object': ch['object'],
                            'last_chord_offset': ch['time_offset'], 'last_chord_duration': ch['duration'],
                            'termination': v['termination'], 'voice_excess': v['excess'],
                            'measure_abnormal': m['abnormal'], 'expected': f['state']['expected'],
                            'actual': f['state']['actual'], 'stack_excess': f['state']['excess'],
                            'peer_measure_excesses': [p['shared_excess'] for p in f['state']['measures']]}
        raise AssertionError('Actual control Voice missing')
    controls.append({'name': c['name'], 'declared_in_memory_control': True,
                     'before': checked(b), 'after': checked(a),
                     'identities_preserved': b['actual_owners_slots_relations_and_manual_status_retained'] and a['actual_owners_slots_relations_and_manual_status_retained']})
assert source['declared_controls']['held_values_and_refs_restored']
summary = {
    'schema': 'audiveris.selected-original-PAGE-owner-replay.v1',
    'source_commit': source['source_commit'], 'observed_sha256': sha(ROOT / 'observed.json'),
    'book_sha256_before_and_after': provenance['book_before_sha256'],
    'fresh_image_recognition': False, 'whole_PAGE_executed': False,
    'whole_MeasureFixer_process_executed': False, 'native_outputs_used': False,
    'qualification': 'Post-PAGE saved-owner method replay. Original initial nullable transient Voice termination/kind/startingStaff differ from live native RHYTHMS; do not clear native values to match them.',
    'replays': rows, 'declared_controls': controls, 'held_values_and_refs_restored': True,
}
